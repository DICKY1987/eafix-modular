// Rate limiting with exponential backoff
class APIRateLimiter {
    private queue: Array<() => Promise<any>> = [];
    private processing = false;
    private delay = 100; // Start with 100ms delay
    
    async execute<T>(apiCall: () => Promise<T>): Promise<T> {
        return new Promise((resolve, reject) => {
            this.queue.push(async () => {
                try {
                    const result = await apiCall();
                    this.delay = Math.max(50, this.delay * 0.9); // Reduce delay on success
                    resolve(result);
                } catch (error) {
                    if (this.isRateLimit(error)) {
                        this.delay = Math.min(5000, this.delay * 2); // Increase delay on rate limit
                    }
                    reject(error);
                }
            });
            
            this.processQueue();
        });
    }
    
    private async processQueue() {
        if (this.processing || this.queue.length === 0) return;
        
        this.processing = true;
        while (this.queue.length > 0) {
            const task = this.queue.shift()!;
            await task();
            await new Promise(resolve => setTimeout(resolve, this.delay));
        }
        this.processing = false;
    }
}

import { exec } from 'child_process';
import { LangChainAgent } from './langchain-agent';

class AgentProcess {
    private agent: LangChainAgent;
    
    constructor() {
        this.agent = new LangChainAgent();
        this.setupIPCHandlers();
    }
    
    private setupIPCHandlers() {
        process.on('message', async (message) => {
            if (message.type === 'execute') {
                // Can execute system commands safely
                const result = await this.executeCommand(message.data.command);
                process.send!({ id: message.id, result });
            }
        });
    }
    
    private async executeCommand(command: string): Promise<string> {
        return new Promise((resolve, reject) => {
            exec(command, (error, stdout, stderr) => {
                if (error) reject(error);
                else resolve(stdout);
            });
        });
    }
}
