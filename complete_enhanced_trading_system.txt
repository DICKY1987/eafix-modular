function Find-PythonCommand {
    $pythonCommands = @("python", "python3", "py")
    
    foreach ($cmd in $pythonCommands) {
        try {
            $version = & $cmd --version 2>$null
            if ($LASTEXITCODE -eq 0) {
                # Extract version number
                $versionNumber = $version -replace "Python ", ""
                $majorVersion = [int]($versionNumber.Split('.')[0])
                $minorVersion = [int]($versionNumber.Split('.')[1])
                
                if (($majorVersion -eq 3 -and $minorVersion -ge 8) -or ($majorVersion -gt 3)) {
                    Write-SetupLog "Found compatible Python: $version using command '$cmd'" "SUCCESS"
                    return @{
                        Command = $cmd
                        Version = $versionNumber
                        Compatible = $true
                    }
                } else {
                    Write-SetupLog "Found Python $version but requires 3.8+ using command '$cmd'" "WARNING"
                }
            }
        } catch {
            # Continue to next command
        }
    }
    
    return @{
        Command = $null
        Version = $null
        Compatible = $false
    }
}

function Invoke-HealthCheck {
    Write-Host "ENHANCED SYSTEM HEALTH CHECK" -ForegroundColor Green
    Write-Host "============================" -ForegroundColor Green

    $python = Find-PythonCommand
    if (-not $python) {
        Write-Host "[FAIL] Python not found" -ForegroundColor Red
        return $false
    }

    $utilsScript = "$BaseDirectory\Python\services\python_utilities.py"
    if (!(Test-Path $utilsScript)) {
        Write-Host "[FAIL] python_utilities.py not found. Run setup first." -ForegroundColor Red
        return $false
    }

    # Check cache first
    $cacheFile = "$CacheDirectory\health_cache.json"
    $useCache = $false
    
    if ((Test-Path $cacheFile) -and (-not $Force)) {
        $cacheAge = (Get-Date) - (Get-Item $cacheFile).LastWriteTime
        if ($cacheAge.TotalSeconds -lt 30) {
            $useCache = $true
            Write-TimestampedMessage "Using cached health data (age: $($cacheAge.TotalSeconds.ToString('F0'))s)" "Gray"
        }
    }

    try {
        if ($useCache) {
            $healthJson = Get-Content $cacheFile -Raw
        } else {
            Write-TimestampedMessage "Performing fresh health check..." "Yellow"
            # Ensure cache directory exists
            if (!(Test-Path $CacheDirectory)) {
                New-Item -ItemType Directory -Path $CacheDirectory -Force | Out-Null
            }
            
            $healthJson = & $python $utilsScript health
            
            # Cache the result
            $healthJson | Out-File -FilePath $cacheFile -Encoding UTF8
        }
        
        $health = $healthJson | ConvertFrom-Json

        # Display results based on the Python script's output
        Write-Host ""
        foreach ($check in $health.checks.PSObject.Properties) {
            $checkName = $check.Name
            $checkStatus = $check.Value.status
            $checkPoints = $check.Value.points
            
            $color = switch ($checkStatus) {
                "OK" { "Green" }
                "WARN" { "Yellow" }
                "ERROR" { "Red" }
                default { "Gray" }
            }
            
            $displayName = $checkName -replace "_", " " | ForEach-Object { (Get-Culture).TextInfo.ToTitleCase($_) }
            
            if ($Detailed -and $check.Value.PSObject.Properties['details']) {
                Write-Host "[$checkStatus] $displayName ($checkPoints pts) - $($check.Value.details)" -ForegroundColor $color
            } else {
                Write-Host "[$checkStatus] $displayName ($checkPoints pts)" -ForegroundColor $color
            }
        }

        $score = $health.score
        $maxScore = $health.max_score
        $status = $health.status

        # Performance information
        if ($health.PSObject.Properties['performance']) {
            $perf = $health.performance
            Write-Host ""
            Write-Host "PERFORMANCE METRICS:" -ForegroundColor Cyan
            Write-Host "Check Duration: $($perf.check_duration_ms)ms" -ForegroundColor Gray
            Write-Host "Cache Status: $($perf.cache_status)" -ForegroundColor Gray
        }

        # Final score
        $scoreColor = switch ($status) {
            "healthy" { "Green" }
            "warning" { "Yellow" }
            default { "Red" }
        }

        Write-Host ""
        Write-Host "HEALTH SCORE: $score/$maxScore" -ForegroundColor $scoreColor

        switch ($status) {
            "healthy" {
                Write-Host "✓ System is healthy and ready for trading!" -ForegroundColor Green
            }
            "warning" {
                Write-Host "⚠ System needs some attention" -ForegroundColor Yellow
            }
            default {
                Write-Host "✗ System requires setup or repair" -ForegroundColor Red
            }
        }

        return $status -ne "critical"
    } catch {
        Write-Host "ERROR during health check: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

        }
        else if (bearish_cross) {
            signal.direction = "SELL";
            signal.confidence = CalculateEnhancedConfidence("SELL", volume_confirmation);
            
            if (signal.confidence < m_risk_params.min_confidence_threshold) {
                return false;
            }
            
            signal.signal_type = "technical";
            signal.risk_level = signal.confidence > 0.8 ? "low" : (signal.confidence > 0.7 ? "medium" : "high");
            signal.metadata = StringFormat("MA20:%.5f,MA50:%.5f,Vol:%.0f", ma_fast_0, ma_slow_0, volume_current);
            
            return WriteEnhancedSignalToFile(signal);
        }
        
        return false;
    }
    
    // Enhanced RSI Signal with divergence detection
    bool GenerateEnhancedRSISignal(SignalData &signal) {
        double rsi_0 = iRSI(NULL, 0, 14, PRICE_CLOSE, 0);
        double rsi_1 = iRSI(NULL, 0, 14, PRICE_CLOSE, 1);
        double rsi_2 = iRSI(NULL, 0, 14, PRICE_CLOSE, 2);
        
        // Enhanced RSI conditions with trend confirmation
        double ma_trend = iMA(NULL, 0, 100, 0, MODE_SMA, PRICE_CLOSE, 0);
        bool uptrend = Close[0] > ma_trend;
        bool downtrend = Close[0] < ma_trend;
        
        // RSI oversold/overbought with trend confirmation
        if (rsi_1 <= 30 && rsi_0 > 30 && uptrend) {
            signal.direction = "BUY";
            signal.confidence = CalculateEnhancedConfidence("BUY", true);
            
            if (signal.confidence < m_risk_params.min_confidence_threshold) {
                return false;
            }
            
            signal.signal_type = "technical";
            signal.risk_level = "medium";
            signal.metadata = StringFormat("RSI:%.2f,Trend:UP", rsi_0);
            
            return WriteEnhancedSignalToFile(signal);
        }
        else if (rsi_1 >= 70 && rsi_0 < 70 && downtrend) {
            signal.direction = "SELL";
            signal.confidence = CalculateEnhancedConfidence("SELL", true);
            
            if (signal.confidence < m_risk_params.min_confidence_threshold) {
                return false;
            }
            
            signal.signal_type = "technical";
            signal.risk_level = "medium";
            signal.metadata = StringFormat("RSI:%.2f,Trend:DOWN", rsi_0);
            
            return WriteEnhancedSignalToFile(signal);
        }
        
        return false;
    }
    
    // Enhanced Support/Resistance Signal
    bool GenerateEnhancedSRSignal(SignalData &signal) {
        double current_price = Close[0];
        
        // Calculate dynamic support and resistance levels
        double high_levels[5], low_levels[5];
        for (int i = 0; i < 5; i++) {
            high_levels[i] = High[i+1];
            low_levels[i] = Low[i+1];
        }
        
        // Find highest high and lowest low
        double resistance = high_levels[ArrayMaximum(high_levels)];
        double support = low_levels[ArrayMinimum(low_levels)];
        
        // Check for breakouts with confirmation
        bool resistance_break = (current_price > resistance) && (Close[1] <= resistance);
        bool support_break = (current_price < support) && (Close[1] >= support);
        
        // Add momentum confirmation
        double momentum = iMomentum(NULL, 0, 14, PRICE_CLOSE, 0);
        bool momentum_bullish = momentum > 100;
        bool momentum_bearish = momentum < 100;
        
        if (resistance_break && momentum_bullish) {
            signal.direction = "BUY";
            signal.confidence = CalculateEnhancedConfidence("BUY", true);
            
            if (signal.confidence < m_risk_params.min_confidence_threshold) {
                return false;
            }
            
            signal.signal_type = "technical";
            signal.risk_level = "high"; // Breakouts are higher risk
            signal.metadata = StringFormat("Resistance:%.5f,Mom:%.2f", resistance, momentum);
            
            return WriteEnhancedSignalToFile(signal);
        }
        else if (support_break && momentum_bearish) {
            signal.direction = "SELL";
            signal.confidence = CalculateEnhancedConfidence("SELL", true);
            
            if (signal.confidence < m_risk_params.min_confidence_threshold) {
                return false;
            }
            
            signal.signal_type = "technical";
            signal.risk_level = "high";
            signal.metadata = StringFormat("Support:%.5f,Mom:%.2f", support, momentum);
            
            return WriteEnhancedSignalToFile(signal);
        }
        
        return false;
    }
    
    // Enhanced confidence calculation with multiple factors
    double CalculateEnhancedConfidence(string direction, bool additional_confirmation = false) {
        double confidence = 0.5; // Base confidence
        
        // Factor 1: Trend strength
        double ma_20 = iMA(NULL, 0, 20, 0, MODE_SMA, PRICE_CLOSE, 0);
        double ma_50 = iMA(NULL, 0, 50, 0, MODE_SMA, PRICE_CLOSE, 0);
        double ma_100 = iMA(NULL, 0, 100, 0, MODE_SMA, PRICE_CLOSE, 0);
        
        if (direction == "BUY") {
            if (ma_20 > ma_50 && ma_50 > ma_100) confidence += 0.2; // Strong uptrend
            else if (ma_20 > ma_50) confidence += 0.1; // Weak uptrend
        } else {
            if (ma_20 < ma_50 && ma_50 < ma_100) confidence += 0.2; // Strong downtrend
            else if (ma_20 < ma_50) confidence += 0.1; // Weak downtrend
        }
        
        // Factor 2: RSI confirmation
        double rsi = iRSI(NULL, 0, 14, PRICE_CLOSE, 0);
        if (direction == "BUY" && rsi < 70 && rsi > 30) {
            confidence += 0.15;
        } else if (direction == "SELL" && rsi > 30 && rsi < 70) {
            confidence += 0.15;
        }
        
        // Factor 3: Additional confirmation (volume, momentum, etc.)
        if (additional_confirmation) {
            confidence += 0.1;
        }
        
        // Factor 4: Market volatility adjustment
        double atr = iATR(NULL, 0, 14, 0);
        double atr_avg = 0;
        for (int i = 1; i <= 20; i++) {
            atr_avg += iATR(NULL, 0, 14, i);
        }
        atr_avg /= 20;
        
        // Reduce confidence in highly volatile markets
        if (atr > atr_avg * 1.5) {
            confidence -= 0.1;
        }
        
        // Ensure confidence stays within bounds
        if (confidence > 1.0) confidence = 1.0;
        if (confidence < 0.0) confidence = 0.0;
        
        return confidence;
    }
    
    // Enhanced signal writing with additional fields
    bool WriteEnhancedSignalToFile(SignalData &signal) {
        string signal_id = signal.symbol + "_" + IntegerToString(TimeCurrent()) + "_" + IntegerToString(GetTickCount());
        
        // Enhanced CSV format with additional fields
        string csv_line = signal_id + "," +
                         signal.symbol + "," +
                         signal.direction + "," +
                         DoubleToString(signal.confidence, 3) + "," +
                         TimeToString(signal.timestamp, TIME_DATE | TIME_SECONDS) + "," +
                         signal.strategy_id + "," +
                         signal.metadata + "," +
                         signal.signal_type + "," +
                         signal.timeframe + "," +
                         signal.risk_level;
        
        int file_handle = FileOpen(m_signal_file, FILE_WRITE | FILE_CSV | FILE_COMMON);
        
        if (file_handle != INVALID_HANDLE) {
            // Move to end of file to append
            FileSeek(file_handle, 0, SEEK_END);
            FileWrite(file_handle, csv_line);
            FileClose(file_handle);
            
            if (m_debug_mode) {
                Print("Enhanced signal written: ", signal.direction, " ", signal.symbol, 
                      " Confidence: ", DoubleToString(signal.confidence, 3),
                      " Strategy: ", signal.strategy_id,
                      " Risk: ", signal.risk_level);
            }
            
            return true;
        } else {
            if (m_debug_mode) {
                Print("ERROR: Could not open signal file for writing: ", GetLastError());
            }
            return false;
        }
    }
    
    // Read responses for performance tracking
    bool ReadResponseFromFile(string signal_id, ResponseData &response) {
        int file_handle = FileOpen(m_response_file, FILE_READ | FILE_CSV | FILE_COMMON);
        
        if (file_handle != INVALID_HANDLE) {
            string line;
            
            // Skip header
            if (FileSize(file_handle) > 0) {
                line = FileReadString(file_handle);
            }
            
            // Read data lines
            while (!FileIsEnding(file_handle)) {
                line = FileReadString(file_handle);
                
                string parts[];
                int count = StringSplit(line, ',', parts);
                
                if (count >= 11 && parts[0] == signal_id) {
                    response.signal_id = parts[0];
                    response.trade_id = parts[1];
                    response.status = parts[2];
                    response.execution_price = StringToDouble(parts[3]);
                    response.timestamp = StringToTime(parts[4]);
                    response.error_message = parts[5];
                    response.lot_size = StringToDouble(parts[6]);
                    response.magic_number = StringToInteger(parts[7]);
                    response.slippage = StringToDouble(parts[8]);
                    response.execution_time_ms = StringToDouble(parts[9]);
                    response.broker_response = parts[10];
                    
                    FileClose(file_handle);
                    return true;
                }
            }
            
            FileClose(file_handle);
        }
        
        return false;
    }
    
    // Get performance statistics
    void PrintPerformanceStats() {
        if (m_debug_mode) {
            Print("=== ENHANCED SIGNAL GENERATOR PERFORMANCE ===");
            Print("MA Strategy - Effectiveness: ", DoubleToString(m_ma_performance.effectiveness, 3),
                  " Signals: ", m_ma_performance.total_signals,
                  " Avg Confidence: ", DoubleToString(m_ma_performance.avg_confidence, 3));
            Print("RSI Strategy - Effectiveness: ", DoubleToString(m_rsi_performance.effectiveness, 3),
                  " Signals: ", m_rsi_performance.total_signals,
                  " Avg Confidence: ", DoubleToString(m_rsi_performance.avg_confidence, 3));
            Print("SR Strategy - Effectiveness: ", DoubleToString(m_sr_performance.effectiveness, 3),
                  " Signals: ", m_sr_performance.total_signals,
                  " Avg Confidence: ", DoubleToString(m_sr_performance.avg_confidence, 3));
            Print("Today's Signals: ", m_signals_today, "/", m_risk_params.max_daily_trades);
            Print("Adaptive Selection: ", m_adaptive_selection ? "Enabled" : "Disabled");
            Print("============================================");
        }
    }

// Global instance for easy access
CEnhancedSignalGenerator* g_enhanced_signal_generator = NULL;

// Initialization function
bool InitializeEnhancedSignalGenerator(string symbol, int magic_number = 12345, bool debug_mode = false) {
    if (g_enhanced_signal_generator != NULL) {
        delete g_enhanced_signal_generator;
    }
    
    g_enhanced_signal_generator = new CEnhancedSignalGenerator(symbol, magic_number, debug_mode, true);
    
    if (debug_mode) {
        Print("Enhanced Signal Generator initialized for ", symbol);
    }
    
    return true;
}

// Cleanup function
void CleanupEnhancedSignalGenerator() {
    if (g_enhanced_signal_generator != NULL) {
        g_enhanced_signal_generator.CleanupOldData();
        g_enhanced_signal_generator.PrintPerformanceStats();
        delete g_enhanced_signal_generator;
        g_enhanced_signal_generator = NULL;
    }
}

// Easy-to-use wrapper functions
bool GenerateEnhancedTradingSignal(string primary_strategy = "MA_Cross", string secondary_strategy = "RSI") {
    if (g_enhanced_signal_generator == NULL) {
        Print("ERROR: Enhanced Signal Generator not initialized");
        return false;
    }
    
    return g_enhanced_signal_generator.GenerateSignal(primary_strategy, secondary_strategy);
}

void ShowEnhancedPerformanceStats() {
    if (g_enhanced_signal_generator != NULL) {
        g_enhanced_signal_generator.PrintPerformanceStats();
    }
}
