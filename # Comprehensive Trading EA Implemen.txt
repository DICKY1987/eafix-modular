## System Architecture Overview

### Master System Design Philosophy
The system is built on three core principles:
1. **Decoupling & Modularity**: Logic separation for independent development
2. **Redundancy & Resilience**: Multiple fail-safes for capital preservation
3. **Flexibility & Control**: External configuration without code recompilation

### High-Level Architecture
```
External Systems (Python/Excel) ←→ Communication Bridge ←→ Master EA ←→ MT4 Terminal
                                           ↓
                                    Database Layer
                                           ↓
                                    Configuration Files
```

// Implementation Details for Bridge Interface
bool CBridgeInterface::Initialize(string symbol, string signal_file, string response_file) {
    m_signal_file_path = signal_file;
    m_response_file_path = response_file;
    m_pipe_name = "\\\\.\\pipe\\HUEY_P_ClaudeCentric_SignalPipe";
    m_tcp_port = 8888;
    m_connection_retries = 0;
    
    // Try primary connection method first
    if (ConnectTCP(m_tcp_port)) {
        m_current_method = COMM_TCP_SOCKET;
        m_status = BRIDGE_CONNECTED;
        return true;
    }
    
    // Fallback to named pipe
    if (ConnectNamedPipe(m_pipe_name)) {
        m_current_method = COMM_NAMED_PIPE;
        m_status = BRIDGE_CONNECTED;
        return true;
    }
    
    // Final fallback to file polling
    if (InitializeFilePolling(m_signal_file_path, m_response_file_path)) {
        m_current_method = COMM_FILE_POLLING;
        m_status = BRIDGE_CONNECTED;
        return true;
    }
    
    m_status = BRIDGE_ERROR;
    return false;
}

// Signal Reading Implementation
SignalData CBridgeInterface::ReadSignal() {
    SignalData signal;
    
    switch(m_current_method) {
        case COMM_TCP_SOCKET:
            return ReadSignalFromSocket();
        case COMM_NAMED_PIPE:
            return ReadSignalFromPipe();
        case COMM_FILE_POLLING:
            return ReadSignalFromFile();
    }
    
    return signal; // Empty signal on failure
}

// Trade Outcome Classification Implementation
int CReentryLogic::ClassifyTradeOutcome(const TradeResult& trade) {
    // Get original trade parameters
    double entry_price = trade.entry_price;
    double exit_price = trade.exit_price;
    
    // Calculate theoretical SL and TP levels
    double stop_loss_level, take_profit_level;
    if (trade.direction == "BUY") {
        // For buy trades, SL is below entry, TP above
        stop_loss_level = entry_price - (m_current_params.stop_loss * Point * 10);
        take_profit_level = entry_price + (m_current_params.take_profit * Point * 10);
    } else {
        // For sell trades, SL is above entry, TP below
        stop_loss_level = entry_price + (m_current_params.stop_loss * Point * 10);
        take_profit_level = entry_price - (m_current_params.take_profit * Point * 10);
    }
    
    double breakeven_level = entry_price;
    
    // Classify based on exit price relative to key levels
    double tolerance = Point * 5; // 5 point tolerance for level matching
    
    // Logic 1: Hit stop loss
    if (MathAbs(exit_price - stop_loss_level) <= tolerance) {
        return 1;
    }
    
    // Logic 2: Between stop loss and breakeven (partial loss)
    if ((trade.direction == "BUY" && exit_price > stop_loss_level && exit_price < breakeven_level) ||
        (trade.direction == "SELL" && exit_price < stop_loss_level && exit_price > breakeven_level)) {
        return 2;
    }
    
    // Logic 3: At breakeven
    if (MathAbs(exit_price - breakeven_level) <= tolerance) {
        return 3;
    }
    
    // Logic 4: Between breakeven and take profit (partial profit)
    if ((trade.direction == "BUY" && exit_price > breakeven_level && exit_price < take_profit_level) ||
        (trade.direction == "SELL" && exit_price < breakeven_level && exit_price > take_profit_level)) {
        return 4;
    }
    
    // Logic 5: At take profit
    if (MathAbs(exit_price - take_profit_level) <= tolerance) {
        return 5;
    }
    
    // Logic 6: Beyond take profit (exceptional profit)
    if ((trade.direction == "BUY" && exit_price > take_profit_level) ||
        (trade.direction == "SELL" && exit_price < take_profit_level)) {
        return 6;
    }
    
    // Default: Classify as breakeven if unclear
    return 3;
}

// Reentry Signal Generation
SignalData CReentryLogic::CreateReentrySignal(const TradeResult& source_trade, int action_number) {
    SignalData reentry_signal;
    ReentryAction action = m_actions[action_number];
    
    // Basic signal properties
    reentry_signal.signal_id = "REENTRY_" + source_trade.trade_id + "_" + IntegerToString(action_number);
    reentry_signal.symbol = source_trade.symbol;
    reentry_signal.signal_type = "REENTRY";
    reentry_signal.source = "REENTRY_LOGIC";
    reentry_signal.signal_time = TimeCurrent();
    reentry_signal.source_trade_id = source_trade.trade_id;
    reentry_signal.reentry_action = action_number;
    
    // Determine direction based on action type
    switch(action.action_type) {
        case ACTION_REVERSE:
            reentry_signal.direction = (source_trade.direction == "BUY") ? "SELL" : "BUY";
            break;
        default:
            reentry_signal.direction = source_trade.direction; // Same direction
            break;
    }
    
    // Calculate lot size adjustment
    reentry_signal.lot_size = source_trade.lot_size * action.size_multiplier;
    
    // Adjust confidence
    reentry_signal.confidence = 0.8 * action.confidence_adjustment; // Base reentry confidence
    
    // Set metadata
    reentry_signal.metadata = StringFormat("Reentry Action %d: %s", 
                                         action_number, 
                                         EnumToString(action.action_type));
    
    return reentry_signal;
}

        Print("Failed to parse calendar CSV file");

    if (!ProcessCalendarData()) {
        Print("Failed to process calendar data");
        return false;
    }
    
    if (m_config.process_anticipation) {
        if (!GenerateAnticipationEvents()) {
            Print("Warning: Failed to generate anticipation events");

    }

    string archive_path = StringFormat("%s\\archive\\%s_%d.csv", 
                                     m_config.downloads_path,
                                     "calendar",
                                     (int)TimeCurrent());
    
    FileCopy(file_path, 0, archive_path, 0);
    FileDelete(file_path, 0);
    
    Print("Calendar import completed successfully. Events loaded: ", m_event_count);
    return true;
}

// File Detection Priority System
bool CCalendarSystem::FindCalendarFile(string& file_path) {
    string search_patterns[] = {
        "ff_calendar*.csv",           // ForexFactory - highest priority
        "*calendar*thisweek*.csv",    // This week calendars
        "*economic*calendar*.csv",    // Economic calendar files
        "*calendar*.csv",             // General calendar files
        "*forex*.csv"                 // Forex calendar files
    };
    
    for (int i = 0; i < ArraySize(search_patterns); i++) {
        string pattern = search_patterns[i];
        
        // Search Downloads folder for files matching pattern
        string file_name;
        long search_handle = FileFindFirst(m_config.downloads_path + "\\" + pattern, file_name);
        
        if (search_handle != INVALID_HANDLE) {
            file_path = m_config.downloads_path + "\\" + file_name;
            FileFindClose(search_handle);
            return true;
        }
    }
    
    return false; // No calendar file found
}

SignalData CCalendarSystem::CreateCalendarSignal(const EconomicEvent& event) {
    SignalData signal;
    
    signal.signal_id = "CAL_" + event.event_id + "_" + IntegerToString((int)TimeCurrent());
    signal.symbol = "EURUSD"; // Default symbol for calendar signals
    signal.signal_type = "CALENDAR";
    signal.source = "CALENDAR_SYSTEM";
    signal.signal_time = TimeCurrent();
    signal.parameter_set_id = event.parameter_set_id;
    
    // Determine direction based on currency and event type
    if (event.country == "USD") {
        signal.direction = (event.impact == "High") ? "SELL" : "BUY"; // USD strength assumption
    } else {
        signal.direction = (event.impact == "High") ? "BUY" : "SELL";
    }
    
    // Set confidence based on impact level
    if (event.impact == "High") {
        signal.confidence = 0.8;
    } else if (event.impact == "Medium") {
        signal.confidence = 0.6;
    } else {
        signal.confidence = 0.4;
    }
    
    // Calculate lot size from parameter set
    ParameterSet params = GetParameterSet(event.parameter_set_id);
    signal.lot_size = params.lot_size;
    
    signal.metadata = StringFormat("Calendar Event: %s (%s %s)", 
                                  event.title, 
                                  event.country, 
                                  event.impact);
    
    return signal;
}

    On Error GoTo ErrorHandler

    Dim errorMsg As String: errorMsg = ""
    
    ' Validate lot size
    Dim lotSize As Double
    lotSize = ws.Range("LotSize_" & setID).Value
    If lotSize <= 0 Or lotSize > 10 Then
        valid = False
        errorMsg = errorMsg & "Invalid lot size (must be 0-10)" & vbCrLf
    End If
    
    ' Validate stop loss
    Dim stopLoss As Integer
    stopLoss = ws.Range("StopLoss_" & setID).Value
    If stopLoss <= 0 Or stopLoss > 500 Then
        valid = False
        errorMsg = errorMsg & "Invalid stop loss (must be 1-500 pips)" & vbCrLf
    End If
    
    ' Validate take profit
    Dim takeProfit As Integer
    takeProfit = ws.Range("TakeProfit_" & setID).Value
    If takeProfit <= 0 Or takeProfit > 1000 Then
        valid = False
        errorMsg = errorMsg & "Invalid take profit (must be 1-1000 pips)" & vbCrLf
    End If
    
    ' Validate risk percentage
    Dim riskPercent As Double
    riskPercent = ws.Range("RiskPercent_" & setID).Value
    If riskPercent <= 0 Or riskPercent > 10 Then
        valid = False
        errorMsg = errorMsg & "Invalid risk percentage (must be 0-10%)" & vbCrLf
    End If
    
    ' Check parameter relationships
    If takeProfit <= stopLoss Then
        valid = False
        errorMsg = errorMsg & "Take profit must be greater than stop loss" & vbCrLf
    End If
    
    If Not valid Then
        MsgBox "Validation errors:" & vbCrLf & errorMsg, vbExclamation
    End If
    
    ValidateParameterSetData = valid
    Exit Function
    
ErrorHandler:
    ValidateParameterSetData = False

// Risk Assessment Implementation
RISK_LEVEL CRiskManager::AssessRiskLevel() {
    UpdateRiskMetrics();
    
    // Emergency level checks
    if (m_metrics.current_drawdown > m_max_drawdown_percent ||
        m_metrics.consecutive_losses >= m_max_consecutive_losses ||
        m_metrics.daily_loss_amount > m_metrics.daily_loss_limit) {
        return RISK_EMERGENCY;
    }
    
    // Critical level checks
    if (m_metrics.current_drawdown > (m_max_drawdown_percent * 0.8) ||
        m_metrics.consecutive_losses >= (m_max_consecutive_losses - 1) ||
        m_metrics.daily_loss_amount > (m_metrics.daily_loss_limit * 0.8)) {
        return RISK_CRITICAL;
    }
    
    // Warning level checks
    if (m_metrics.current_drawdown > (m_max_drawdown_percent * 0.6) ||
        m_metrics.consecutive_losses >= (m_max_consecutive_losses - 2) ||
        m_metrics.daily_loss_amount > (m_metrics.daily_loss_limit * 0.6)) {
        return RISK_WARNING;
    }
    
    return RISK_NORMAL;
}

// Trade Permission Checking
bool CRiskManager::IsTradeAllowed(const SignalData& signal) {
    // Check if emergency halt is active
    if (m_emergency_halt_active) {
        return false;
    }
    
    // Check global halt flag
    if (GlobalVariableGet("GVAR_TRADE_HALT") > 0) {
        return false;
    }
    
    // Check risk level restrictions
    RISK_LEVEL current_level = AssessRiskLevel();
    
    switch(current_level) {
        case RISK_EMERGENCY:
            return false; // No new trades allowed
            
        case RISK_CRITICAL:
            // Only allow high confidence trades
            return (signal.confidence >= 0.9);
            
        case RISK_WARNING:
            // Only allow medium-high confidence trades
            return (signal.confidence >= 0.7);
            
        case RISK_NORMAL:
            // Normal trading allowed
            return true;
    }
    
    return false;
}

// Watchdog Check Implementation
void CSystemWatchdog::PerformWatchdogCheck() {
    if (!m_watchdog_enabled) return;
    
    datetime current_time = TimeCurrent();
    if (current_time - m_last_check < m_check_interval_seconds) return;
    
    // Update peak equity tracking
    double current_equity = AccountEquity();
    if (current_equity > m_peak_equity) {

    // Perform safety checks

        TriggerEmergencyShutdown("Account safety check failed");
        return;
    }
    
    if (!CheckSystemHealth()) {
        TriggerEmergencyShutdown("System health check failed");
        return;
    }
    
    m_last_check = current_time;
    
    // Update global variables for EA monitoring
    GlobalVariableSet("GVAR_PEAK_EQUITY", m_peak_equity);
    GlobalVariableSet("GVAR_CURRENT_DRAWDOWN", GetCurrentDrawdown());
    GlobalVariableSet("GVAR_CONSECUTIVE_LOSSES", GetConsecutiveLossCount());

// Performance Report Generation
string CAnalyticsEngine::GeneratePerformanceReport(datetime start_date, datetime end_date) {
    string report = "TRADING SYSTEM PERFORMANCE REPORT\n";
    report += "=====================================\n\n";
    
    // Basic Statistics
    int total_trades = GetTradeCount(start_date, end_date);
    int winning_trades = GetWinningTradeCount(start_date, end_date);
    int losing_trades = total_trades - winning_trades;
    
    report += StringFormat("Period: %s to %s\n", 
                          TimeToStr(start_date, TIME_DATE),
                          TimeToStr(end_date, TIME_DATE));
    report += StringFormat("Total Trades: %d\n", total_trades);
    report += StringFormat("Winning Trades: %d (%.1f%%)\n", 
                          winning_trades, 
                          (double)winning_trades / total_trades * 100);
    report += StringFormat("Losing Trades: %d (%.1f%%)\n", 
                          losing_trades, 
                          (double)losing_trades / total_trades * 100);
    
    // Profitability Metrics
    double total_profit = GetTotalProfit(start_date, end_date);
    double total_loss = GetTotalLoss(start_date, end_date);
    double net_profit = total_profit + total_loss;
    double profit_factor = (total_loss != 0) ? (total_profit / MathAbs(total_loss)) : 0;
    
    report += StringFormat("\nProfitability:\n");
    report += StringFormat("Total Profit: %.2f\n", total_profit);
    report += StringFormat("Total Loss: %.2f\n", total_loss);
    report += StringFormat("Net Profit: %.2f\n", net_profit);
    report += StringFormat("Profit Factor: %.2f\n", profit_factor);
    
    // Parameter Set Performance
    report += "\nParameter Set Performance:\n";
    for (int i = 1; i <= 10; i++) {
        double set_profit = GetParameterSetProfit(i, start_date, end_date);
        int set_trades = GetParameterSetTradeCount(i, start_date, end_date);
        report += StringFormat("Set %d: %.2f (% d trades)\n", i, set_profit, set_trades);
    }
    
    // Reentry Performance
    report += "\nReentry System Performance:\n";
    for (int action = 1; action <= 6; action++) {
        double action_profit = GetReentryActionProfit(action, start_date, end_date);
        int action_count = GetReentryActionCount(action, start_date, end_date);
        report += StringFormat("Action %d: %.2f (%d executions)\n", 
                              action, action_profit, action_count);
    }
    
    return report;
}
