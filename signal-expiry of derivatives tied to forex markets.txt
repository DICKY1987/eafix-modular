import requests
from bs4 import BeautifulSoup
import re
import csv
from datetime import datetime

def get_forexlive_expiries():
    url = "https://www.forexlive.com/technical-analysis/"
    headers = {'User-Agent': 'Mozilla/5.0'}
    response = requests.get(url, headers=headers)
    soup = BeautifulSoup(response.text, 'html.parser')

    # Search for post titles containing 'option expiries'
    expiry_posts = soup.find_all('a', href=True, text=re.compile("option expiries", re.I))
    if not expiry_posts:
        return []

    # Open the latest post
    expiry_url = expiry_posts[0]['href']
    post_response = requests.get(expiry_url, headers=headers)
    post_soup = BeautifulSoup(post_response.text, 'html.parser')

    text_blocks = post_soup.get_text()
    lines = text_blocks.split('\n')

    results = []
    for line in lines:
        if re.search(r'\b[A-Z]{3}/[A-Z]{3}\b', line):
            match = re.findall(r'([A-Z]{3}/[A-Z]{3}):\s*(.*)', line)
            if match:
                pair, details = match[0]
                strike_data = re.findall(r'(\d+\.\d+|\d+)\s*\(([^)]+)\)', details)
                for strike, size in strike_data:
                    results.append({
                        "pair": pair,
                        "strike": float(strike),
                        "size": size,
                        "source": expiry_url,
                        "timestamp": datetime.utcnow().isoformat()
                    })
    return results

import requests
from bs4 import BeautifulSoup
import re
from forex_python.converter import CurrencyRates
from openpyxl import Workbook
from openpyxl.styles import PatternFill, Font

def get_live_spot(pair):
    c = CurrencyRates()
    base, quote = pair.split('/')
    try:
        rate = c.get_rate(base, quote)
        return rate
    except:
        return None

def parse_forexlive_expiries():
    url = "https://www.forexlive.com/technical-analysis/"
    headers = {'User-Agent': 'Mozilla/5.0'}
    response = requests.get(url, headers=headers)
    soup = BeautifulSoup(response.text, 'html.parser')
    expiry_links = soup.find_all('a', href=True, text=re.compile("option expiries", re.I))
    if not expiry_links:
        return []

    expiry_url = expiry_links[0]['href']
    post_html = requests.get(expiry_url, headers=headers).text
    post_text = BeautifulSoup(post_html, 'html.parser').get_text()

    lines = post_text.split('\n')
    expiries = []

    for line in lines:
        if re.search(r'\b[A-Z]{3}/[A-Z]{3}\b', line):
            match = re.findall(r'([A-Z]{3}/[A-Z]{3}):\s*(.*)', line)
            if match:
                pair, details = match[0]
                strike_data = re.findall(r'(\d+\.\d+|\d+)\s*\(([^)]+)\)', details)
                spot = get_live_spot(pair)
                for strike, size in strike_data:
                    strike = float(strike)
                    pips = round((strike - spot) * (10000 if 'JPY' not in pair else 100), 1) if spot else None
                    notional_numeric = float(re.sub(r"[^\d.]", "", size)) if re.search(r"\d", size) else 0
                    currency = size.split()[0] if re.search(r"[A-Z]{3}", size) else "UNKNOWN"
                    expiries.append({
                        "Pair": pair,
                        "Strike": strike,
                        "Notional": size,
                        "Spot": round(spot, 5) if spot else "N/A",
                        "Pips From Spot": pips,
                        "Risk Zone": "YES" if spot and abs(pips) < 25 and notional_numeric >= 500 else "NO"
                    })
    return expiries

def export_to_excel(data, filename="FX_Option_Expiries.xlsx"):
    wb = Workbook()
    ws = wb.active
    ws.title = "Expiries"

    headers = list(data[0].keys())
    ws.append(headers)

    for row in data:
        ws.append(list(row.values()))

    # Style headers
    for col in range(1, len(headers)+1):
        cell = ws.cell(row=1, column=col)
        cell.font = Font(bold=True)
        cell.fill = PatternFill(start_color="DDDDDD", end_color="DDDDDD", fill_type="solid")

    # Apply conditional formatting for "Risk Zone"
    for row_idx in range(2, len(data)+2):
        cell = ws.cell(row=row_idx, column=headers.index("Risk Zone") + 1)
        if cell.value == "YES":
            for col_idx in range(1, len(headers)+1):
                ws.cell(row=row_idx, column=col_idx).fill = PatternFill(start_color="FF9999", end_color="FF9999", fill_type="solid")

    wb.save(filename)
    print(f"Exported to {filename}")

Python Script  ➜  Excel Sheet ("Signal_Schedule")  ➜  VBA Import Script  ➜  Signal Filtering  ➜  MT4 Execution
Your MT4 system reads signals from a structured sheet (Signal_Schedule) and validates them based on time, symbol, and status before executing.

✅ GOAL
Take the auto-generated signals from the expiry tracker and:

Insert them into the Signal_Schedule sheet

Let your MT4 system read and act on them like any other scheduled signal

🧩 WHERE THIS FITS IN YOUR SYSTEM
📍 Layer 1: Scheduling Layer
Uses signal_schedule_sheet.vb to read new rows

Our script generates those rows from expiry proximity logic

📍 Layer 2: Signal Processing Layer
Your system checks if the current time matches a Trigger Time

Filters based on symbol, state, and priority rules

🛠 STEP-BY-STEP INTEGRATION
🟢 STEP 1: File Naming and Placement
Save/export the expiry tracker as FX_Option_Expiry_Dashboard.xlsx

Place it in the same folder where your main MT4 Excel workbook resides

✅ Tip: Use relative file paths in your VBA to keep portability intact.

🟢 STEP 2: Open the Main Excel Workbook
You’ll either:

Add a new tab called Signal_Schedule and reference this sheet

Or link the existing Signal_Schedule tab to the one in the expiry file

Option A: LINK sheet (Recommended)
In your master workbook:

Add a new sheet called Signal_Schedule_Expiry

In cell A1, type:

excel
Copy
Edit
='[FX_Option_Expiry_Dashboard.xlsx]Signal_Schedule'!A1
Drag to auto-populate columns A–F

🟢 STEP 3: Update VBA Signal Scheduler (if needed)
In signal_schedule_sheet.vb or equivalent, ensure it:

Accepts the new sheet (Signal_Schedule_Expiry) as a valid source

Filters for Strategy = FX_EXPIRY to distinguish expiry-based signals

You can add:

vb
Copy
Edit
If SheetName = "Signal_Schedule_Expiry" Then
    StrategyType = "FX_EXPIRY"
End If
🟢 STEP 4: Use Existing Signal Processing Logic
Once the row hits Signal_Schedule:

It will be processed like any other scheduled signal

Triggered based on its Trigger Time and conditions

Trade will be executed via your existing MT4 VBA bridge
