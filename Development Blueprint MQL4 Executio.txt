      void OnTick() {
          // Update current market data
          RefreshRates();
          
          // Check circuit breaker conditions
          if(g_RiskManager.IsCircuitBreakerActive()) {
              if(g_StateManager.GetCurrentState() != STATE_PAUSED) {
                  g_StateManager.SetState(STATE_PAUSED);
                  g_Logger.LogWarning("Circuit breaker activated - trading paused");
              }
              return;
          }
          
          // Process based on current state
          switch(g_StateManager.GetCurrentState()) {
              case STATE_IDLE:
                  ProcessIdleState();
                  break;
                  
              case STATE_ORDERS_PLACED:
                  ProcessOrdersPlacedState();
                  break;
                  
              case STATE_TRADE_TRIGGERED:
                  ProcessTradeTriggeredState();
                  break;
                  
              case STATE_PAUSED:
                  ProcessPausedState();
                  break;
                  
              case STATE_ERROR:
                  ProcessErrorState();
                  break;
          }
          
          // Monitor existing positions
          MonitorActivePositions();
          
          // Update performance metrics
          g_RiskManager.UpdateMetrics();
      }
      
      void ProcessIdleState() {
          // Check for new signals
          SignalData signal;
          if(g_SignalProcessor.GetNextSignal(signal)) {
              if(ValidateSignal(signal)) {
                  ExecuteSignal(signal);
              }
          }
      }
      
      void ExecuteSignal(const SignalData& signal) {
          // Calculate position size
          double lotSize = g_RiskManager.CalculatePositionSize(
              signal.stopLoss, 
              signal.lotSize
          );
          
          // Place straddle orders
          int buyTicket = PlaceStopOrder(OP_BUYSTOP, lotSize, signal);
          int sellTicket = PlaceStopOrder(OP_SELLSTOP, lotSize, signal);
          
          if(buyTicket > 0 && sellTicket > 0) {
              g_StateManager.SetState(STATE_ORDERS_PLACED);
              g_StateManager.SetPendingOrders(buyTicket, sellTicket);
              g_Logger.LogInfo("Straddle orders placed successfully");
          }
      }

      void ProcessClosedTrade(const TradeResult& result) {
          // Calculate R-value
          double R = CalculateRValue(result);
          
          // Get money loss/gain
          double ML = result.profit < 0 ? MathAbs(result.profit) : 0;
          double MG = result.profit > 0 ? result.profit : 0;
          
          // Get current balance
          double B = AccountBalance();
          
          // Determine die roll outcome (1-6)
          int dieRoll = g_ReentryLogic.DetermineOutcome(R, ML, MG, B);
          
          g_Logger.LogInfo(StringFormat(
              "Trade closed - Die Roll: %d, R: %.2f, P/L: %.2f",
              dieRoll, R, result.profit
          ));
          
          // Execute reentry action if enabled
          if(EnableReentrySystem && dieRoll > 0) {
              ReentryAction action = g_ReentryLogic.GetAction(dieRoll);
              ExecuteReentryAction(action, result);
          }
          
          // Update state
          g_StateManager.UpdateTradeHistory(result);
          g_StateManager.SetState(STATE_IDLE);
      }
      
      void ExecuteReentryAction(const ReentryAction& action, const TradeResult& original) {
          switch(action.type) {
              case REENTRY_NO_ACTION:
                  g_Logger.LogInfo("No reentry action - waiting for next signal");
                  break;
                  
              case REENTRY_WAIT_SIGNAL:
                  g_StateManager.SetWaitUntil(TimeCurrent() + action.delaySeconds);
                  break;
                  
              case REENTRY_SAME_TRADE:
                  Sleep(action.delaySeconds * 1000);
                  PlaceReentryTrade(original.direction, action.sizeMultiplier);
                  break;
                  
              case REENTRY_REVERSE:
                  Sleep(action.delaySeconds * 1000);
                  int reverseDir = original.direction == OP_BUY ? OP_SELL : OP_BUY;
                  PlaceReentryTrade(reverseDir, action.sizeMultiplier);
                  break;
                  
              case REENTRY_REDUCE_SIZE:
                  g_RiskManager.AdjustRiskPercent(action.confidenceAdjustment);
                  break;
                  
              case REENTRY_INCREASE_SIZE:
                  g_RiskManager.AdjustRiskPercent(action.confidenceAdjustment);
                  PlaceReentryTrade(original.direction, action.sizeMultiplier);
                  break;
          }
      }

      // Error code definitions
      #define ERR_CONFIGURATION_LOAD    5001
      #define ERR_SIGNAL_PARSE         5002
      #define ERR_ORDER_PLACEMENT      5003
      #define ERR_STATE_CORRUPTION     5004
      #define ERR_RISK_LIMIT_EXCEEDED  5005
      
      bool HandleError(int errorCode, string context) {
          g_Logger.LogError(StringFormat("Error %d in %s: %s", 
              errorCode, context, ErrorDescription(errorCode)));
          
          switch(errorCode) {
              case ERR_TRADE_TIMEOUT:
              case ERR_REQUOTE:
              case ERR_PRICE_CHANGED:
                  return true; // Recoverable - retry
                  
              case ERR_MARKET_CLOSED:
              case ERR_TRADE_DISABLED:
                  g_StateManager.SetState(STATE_PAUSED);
                  return false;
                  
              case ERR_NOT_ENOUGH_MONEY:
              case ERR_RISK_LIMIT_EXCEEDED:
                  g_RiskManager.ActivateCircuitBreaker();
                  return false;
                  
              default:
                  if(errorCode >= 5000) { // Custom errors
                      g_StateManager.SetState(STATE_ERROR);
                  }
                  return false;
          }
      }
      
      void OnDeinit(const int reason) {
          // Save state before shutdown
          g_StateManager.SaveState();
          
          // Close any pending orders
          CloseAllPendingOrders();
          
          // Final logging
          g_Logger.LogInfo("EA Shutdown - Reason: " + GetUninitReasonText(reason));
          
          // Cleanup
          EventKillTimer();
      }

      int CReentryLogic::DetermineOutcome(double R, double ML, double MG, double B) {
          // Validate inputs
          if(!m_enabled) return 0;
          
          // Die Roll Logic based on R value
          int dieRoll = 0;
          
          if(R <= -1.0) {
              // Stop loss hit or worse
              dieRoll = 1;
          }
          else if(R > -1.0 && R <= -0.5) {
              // Partial loss (50% to stop loss)
              dieRoll = 2;
          }
          else if(R > -0.5 && R < 0.5) {
              // Breakeven range (-50% to +50% of risk)
              dieRoll = 3;
          }
          else if(R >= 0.5 && R < 1.0) {
              // Partial profit (50% to take profit)
              dieRoll = 4;
          }
          else if(R >= 1.0 && R < 1.5) {
              // Take profit hit
              dieRoll = 5;
          }
          else if(R >= 1.5) {
              // Beyond take profit
              dieRoll = 6;
          }
          
          // Log the outcome
          LogOutcome(dieRoll, R, ML, MG, B);
          
          // Update statistics
          UpdateStatistics(dieRoll, MG - ML);
          
          return dieRoll;
      }
      
      double CReentryLogic::CalculateRValue(const TradeResult &result) {
          double risk = MathAbs(result.openPrice - result.stopLoss);
          if(risk == 0) return 0;
          
          double profit = result.closePrice - result.openPrice;
          if(result.direction == OP_SELL) {
              profit = result.openPrice - result.closePrice;
          }
          
          return profit / risk;
      }

      bool CReentryLogic::LoadProfile(string filePath) {
          // Open CSV file
          int fileHandle = FileOpen(filePath, FILE_READ|FILE_CSV|FILE_ANSI, ',');
          if(fileHandle == INVALID_HANDLE) {
              Print("Failed to open reentry profile: ", filePath);
              return false;
          }
          
          // Skip header row
          if(!FileReadString(fileHandle)) {
              FileClose(fileHandle);
              return false;
          }
          
          // Read action configurations
          while(!FileIsEnding(fileHandle)) {
              string line = FileReadString(fileHandle);
              if(StringLen(line) == 0) continue;
              
              // Parse CSV line
              string fields[];
              int fieldCount = StringSplit(line, ',', fields);
              if(fieldCount < 6) continue;
              
              int actionNum = (int)StringToInteger(fields[0]);
              if(actionNum < 1 || actionNum > 6) continue;
              
              // Populate action structure
              m_actions[actionNum].actionNumber = actionNum;
              m_actions[actionNum].type = StringToReentryType(fields[1]);
              m_actions[actionNum].sizeMultiplier = StringToDouble(fields[2]);
              m_actions[actionNum].delaySeconds = (int)StringToInteger(fields[3]);
              m_actions[actionNum].confidenceAdjustment = StringToDouble(fields[4]);
              m_actions[actionNum].parameters = fields[5];
          }
          
          FileClose(fileHandle);
          m_profileName = filePath;
          Print("Loaded reentry profile: ", m_profileName);
          return true;
      }

      Public Function ImportCalendarData(sourcePath As String) As Boolean
          Dim ws As Worksheet
          Dim lastRow As Long
          Dim i As Long
          Dim evt As clsEconomicEvent
          
          On Error GoTo ErrorHandler
          
          ' Clear existing events
          Set g_Events = New Collection

          Workbooks.Open sourcePath, ReadOnly:=True
          Set ws = ActiveWorkbook.Sheets(1)
          
          ' Find last row
          lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
          
          ' Import events (skip header)
          For i = 2 To lastRow
              Set evt = New clsEconomicEvent
              
              With evt
                  .EventDate = CDate(ws.Cells(i, 1).Value)
                  .EventTime = TimeValue(ws.Cells(i, 2).Value)
                  .Currency = UCase(Trim(ws.Cells(i, 3).Value))
                  .EventName = Trim(ws.Cells(i, 4).Value)
                  .Impact = UCase(Trim(ws.Cells(i, 5).Value))
                  .Previous = ws.Cells(i, 6).Value
                  .Forecast = ws.Cells(i, 7).Value
                  .EventID = GenerateEventID(evt)
                  
                  ' Validate event
                  If .Validate() Then
                      g_Events.Add evt, .EventID
                  End If
              End With
          Next i
          
          ' Close source file
          ActiveWorkbook.Close SaveChanges:=False
          
          Debug.Print "Imported " & g_Events.Count & " events"
          ImportCalendarData = True
          Exit Function
          
      ErrorHandler:
          LogError "ImportCalendarData", Err.Description
          ImportCalendarData = False
      End Function

      '=== modSignalGeneration ===
      Public Sub GenerateTradingSignals()
          Dim evt As clsEconomicEvent
          Dim sig As clsSignal
          Dim symbols As Variant
          Dim i As Integer
          
          On Error GoTo ErrorHandler
          
          ' Process each event
          For Each evt In g_Events
              ' Check if event needs signal
              If ShouldGenerateSignal(evt) Then
                  ' Get affected symbols
                  symbols = GetAffectedSymbols(evt.Currency)
                  
                  ' Create signal for each symbol
                  For i = 0 To UBound(symbols)
                      Set sig = New clsSignal
                      
                      With sig
                          .Symbol = symbols(i)
                          .EventID = evt.EventID
                          .SignalTime = DateAdd("n", -g_Config.MinutesBeforeEvent, _
                                              evt.EventDate + evt.EventTime)
                          .OrderType = "both"  ' Straddle
                          .LotSize = CalculateLotSize(evt.Impact)
                          .StopLoss = g_Config.GetStopLoss(evt.Impact)
                          .TakeProfit = g_Config.GetTakeProfit(evt.Impact)
                          .MagicNumber = g_Config.BaseMagicNumber + i
                          .Comment = evt.Currency & "_" & evt.EventName & "_" & _
                                    Format(evt.EventDate, "YYYYMMDD")
                          .Expires = g_Config.SignalExpirySeconds
                          
                          ' Generate JSON file
                          If .SaveToFile(SIGNAL_PATH) Then
                              g_ActiveSignals.Add sig, .SignalID
                              LogSignal sig
                          End If
                      End With
                  Next i
              End If
          Next evt
          
          Debug.Print "Generated " & g_ActiveSignals.Count & " signals"
          Exit Sub
          
      ErrorHandler:
          LogError "GenerateTradingSignals", Err.Description
      End Sub

                                           "EURCHF", "EURAUD")
              Case "GBP"
                  GetAffectedSymbols = Array("GBPUSD", "EURGBP", "GBPJPY", _
                                           "GBPCHF", "GBPAUD")
              Case Else
                  GetAffectedSymbols = Array() ' No symbols
          End Select
      End Function

      // FileMonitor.cpp
      #include "FileMonitor.h"
      #include <boost/filesystem.hpp>
      #include <chrono>
      
      namespace fs = boost::filesystem;
      
      void FileMonitor::MonitoringLoop() {
          std::unordered_map<std::string, std::time_t> fileTimestamps;
          
          while (m_running.load()) {
              try {
                  // Scan directory
                  for (const auto& entry : fs::directory_iterator(m_watchPath)) {
                      if (fs::is_regular_file(entry.path()) &&
                          entry.path().extension() == ".json") {
                          
                          std::string filePath = entry.path().string();
                          std::time_t modTime = fs::last_write_time(entry.path());
                          
                          // Check if new or modified
                          auto it = fileTimestamps.find(filePath);
                          if (it == fileTimestamps.end() || it->second < modTime) {
                              fileTimestamps[filePath] = modTime;
                              
                              // Process new signal file
                              if (m_callback) {
                                  m_callback(filePath);
                              }
                              
                              // Move to processed folder
                              MoveToProcessed(filePath);
                          }
                      }
                  }
              }
              catch (const std::exception& e) {
                  LOG_ERROR("File monitoring error: " + std::string(e.what()));
              }
              
              // Sleep for polling interval
              std::this_thread::sleep_for(
                  std::chrono::milliseconds(m_pollingIntervalMs));
          }
      }
